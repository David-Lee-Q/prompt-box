### 导出逻辑 — `src/utils/agent-export.ts`

```typescript
import type { Prompt, Version } from '@/types';

export interface ExportOptions {
  format: AgentExportFormat;
  prompt: Prompt;
  version?: Version;        // 指定版本，否则用当前版本
  variables?: string[];     // 已提取的变量列表
  includeHandler?: boolean; // 是否包含占位 handler
}

export function exportAsAgentTool(options: ExportOptions): AgentExportResult {
  const { format, prompt, version, variables, includeHandler } = options;
  const content = version?.content ?? prompt.content;
  const name = toCamelCase(prompt.name);
  const description = prompt.notes || `Prompt: ${prompt.name}`;

  switch (format) {
    case 'openai-fc':
      return exportOpenAIFunctionCalling(name, description, content, variables, includeHandler);
    case 'anthropic-tools':
      return exportAnthropicTool(name, description, content, variables);
    case 'openai-sdk':
      return exportOpenAISDK(name, description, content, variables);
    case 'langchain':
      return exportLangChain(name, description, content, variables);
  }
}

function exportOpenAIFunctionCalling(
  name: string, description: string, content: string,
  variables?: string[], includeHandler?: boolean
): AgentExportResult {
  const properties: Record<string, unknown> = {};
  const required: string[] = [];

  if (variables) {
    for (const v of variables) {
      properties[v] = { type: 'string', description: `变量 ${v}` };
      required.push(v);
    }
  }

  // 没有变量时，使用 content 作为默认参数
  if (required.length === 0) {
    properties.content = { type: 'string', description: 'Prompt 内容' };
    required.push('content');
  }

  const config: AgentToolConfig = {
    type: 'function',
    function: {
      name,
      description: truncate(description, 200),
      parameters: {
        type: 'object',
        properties,
        required,
        additionalProperties: false,
      },
    },
  };

  // 可选：包含 handler 占位
  if (includeHandler) {
    config.function.handler = `// 使用 ${name} Prompt:\n// ${content.split('\n').join('\n// ')}`;
  }

  return {
    format: 'openai-fc',
    content: JSON.stringify(config, null, 2),
    filename: `${name}-tool.json`,
  };
}

function exportAnthropicTool(
  name: string, description: string, content: string,
  variables?: string[]
): AgentExportResult {
  const properties: Record<string, unknown> = {};
  const required: string[] = [];

  if (variables) {
    for (const v of variables) {
      properties[v] = { type: 'string', description: `变量 ${v}` };
      required.push(v);
    }
  }
  if (required.length === 0) {
    properties.content = { type: 'string', description: 'Prompt 内容' };
    required.push('content');
  }

  const config: AnthropicToolConfig = {
    name,
    description: truncate(description, 200),
    input_schema: {
      type: 'object',
      properties,
      required,
    },
  };

  return {
    format: 'anthropic-tools',
    content: JSON.stringify(config, null, 2),
    filename: `${name}-anthropic-tool.json`,
  };
}

function exportOpenAISDK(
  name: string, description: string, content: string,
  variables?: string[]
): AgentExportResult {
  // 生成 OpenAI SDK tool definition + usage example
  const code = `import OpenAI from 'openai';

const openai = new OpenAI();

const ${name}Tool = {
  type: 'function',
  function: {
    name: '${name}',
    description: ${JSON.stringify(truncate(description, 200))},
    parameters: {
      type: 'object',
      properties: {
        ${(variables ?? ['content']).map((v) =>
          `${v}: { type: 'string', description: '${v}' }`
        ).join(',\n        ')}
      },
      required: [${(variables ?? ['content']).map((v) => `'${v}'`).join(', ')}],
    },
  },
};

// Prompt template:
// \`\`\`
// ${truncate(content, 500)}
// \`\`\`
`;

  return { format: 'openai-sdk', content: code, filename: `${name}.ts` };
}

function exportLangChain(
  name: string, description: string, content: string,
  variables?: string[]
): AgentExportResult {
  const code = `import { StructuredTool } from 'langchain/tools';
import { z } from 'zod';

export class ${pascalCase(name)}Tool extends StructuredTool {
  name = '${name}';
  description = ${JSON.stringify(truncate(description, 200))};

  schema = z.object({
    ${(variables ?? ['content']).map((v) =>
      `${v}: z.string().describe('${v}')`
    ).join(',\n    ')}
  });

  async _call(input: z.infer<typeof this.schema>) {
    // Prompt template:
    // ${truncate(content, 300).split('\n').join('\n    // ')}
    throw new Error('Not implemented');
  }
}
`;

  return { format: 'langchain', content: code, filename: `${name}-langchain-tool.ts` };
}

// 工具函数
function toCamelCase(name: string): string {
  return name
    .replace(/[^a-zA-Z0-9一-鿿\s]/g, '')
    .split(/\s+/)
    .map((word, i) =>
      i === 0
        ? word.toLowerCase()
        : word.charAt(0).toUpperCase() + word.slice(1).toLowerCase()
    )
    .join('')
    .replace(/[一-鿿]/g, '_');
}

function pascalCase(name: string): string {
  const cc = toCamelCase(name);
  return cc.charAt(0).toUpperCase() + cc.slice(1);
}

function truncate(str: string, max: number): string {
  return str.length > max ? str.slice(0, max) + '...' : str;
}

### AI 搜索 Service — `src/services/ai/search.ts`

```typescript
import { getCurrentProvider } from './index';
import { searchPrompts, getAllPrompts } from '@/services/promptService';
import type { Prompt } from '@/types';

export interface AISearchResult {
  prompt: Prompt;
  reason: string;          // 为什么推荐这个结果
  matchType: 'keyword' | 'semantic' | 'related';
}

/**
 * AI 增强搜索：先关键词搜索，再用 AI 扩展语义相关结果
 */
export async function aiEnhancedSearch(query: string): Promise<{
  keywordResults: Prompt[];
  aiSuggestions: AISearchResult[];
}> {
  // 1. 先跑关键词搜索（现有逻辑）
  const keywordResults = await searchPrompts(query);

  // 2. 如果关键词结果足够多，且搜索词很短，不触发 AI
  if (keywordResults.length >= 10 || query.trim().length < 4) {
    return { keywordResults, aiSuggestions: [] };
  }

  // 3. 异步触发 AI 搜索增强（非阻塞）
  try {
    const aiSuggestions = await semanticSearch(query, keywordResults);
    return { keywordResults, aiSuggestions };
  } catch {
    return { keywordResults, aiSuggestions: [] };
  }
}

async function semanticSearch(
  query: string,
  existingResults: Prompt[]
): Promise<AISearchResult[]> {
  const provider = getCurrentProvider();
  if (!provider) return [];

  // 获取已有结果 ID 集合（排除重复）
  const existingIds = new Set(existingResults.map((p) => p.id));

  // 获取全量 Prompt（仅在必要时）
  const allPrompts = await getAllPrompts();
  const candidates = allPrompts.filter((p) => !existingIds.has(p.id));

  // 如果候选太少，不调用 AI
  if (candidates.length === 0 || candidates.length > 50) return [];
  // 限制候选数防止 token 超限
  const limited = candidates.slice(0, 20);

  const promptList = limited
    .map((p) => `ID: ${p.id}\n名称: ${p.name}\n内容: ${p.content.slice(0, 300)}\n标签: ${p.tags.join(', ')}`)
    .join('\n---\n');

  const systemPrompt = `你是一个搜索专家。用户搜索了："${query}"。
以下是候选 Prompt 列表。请选择 1-3 个与用户搜索意图最相关的 Prompt（不包括已在关键词结果中的）。
对每个选中的 Prompt，说明为什么它相关。
返回格式：每行 "ID|推荐理由"`;

  const userPrompt = `候选 Prompt：\n${promptList}`;

  const result = await provider.chat([
    { role: 'system', content: systemPrompt },
    { role: 'user', content: userPrompt },
  ]);

  return parseAISearchResults(result, limited);
}

function parseAISearchResults(text: string, prompts: Prompt[]): AISearchResult[] {
  return text
    .split('\n')
    .filter((line) => line.includes('|'))
    .map((line) => {
      const [id, ...reason] = line.split('|');
      const prompt = prompts.find((p) => p.id === id.trim());
      if (!prompt) return null;
      return {
        prompt,
        reason: reason.join('|').trim(),
        matchType: 'semantic' as const,
      };
    })
    .filter((r): r is AISearchResult => r !== null)
    .slice(0, 3);
}
```

```typescript
// 防抖：搜索暂停 1.5 秒后再触发 AI 搜索
// 避免每次按键都调 AI API（浪费 token）

// Header.tsx 中搜索处理逻辑
const searchTimeoutRef = useRef<ReturnType<typeof setTimeout>>();
const [aiSuggestions, setAiSuggestions] = useState<AISearchResult[]>([]);

const handleSearchInput = (query: string) => {
  setSearchQuery(query);
  clearTimeout(searchTimeoutRef.current);

  // 立即触发关键词搜索（现有行为）
  setTimeout(() => loadPrompts(), 0);

  // 延迟触发 AI 搜索增强
  if (query.trim().length >= 4) {
    searchTimeoutRef.current = setTimeout(async () => {
      const result = await aiEnhancedSearch(query);
      setAiSuggestions(result.aiSuggestions);
    }, 1500);
  } else {
    setAiSuggestions([]);
  }
};
```

### 数据流

```
用户在搜索框输入
       ↓
0ms:   关键词搜索（现有逻辑，立即执行）
       ↓
1500ms: AI 搜索增强（用户在搜索暂停后触发）
  → 获取全量 Prompt
  → 排除已在关键词结果中的
  → 截取前 20 条候选
  → 调用 AI API 理解搜索意图
  → AI 返回 1-3 个语义相关结果 + 推荐理由
       ↓
关键词结果正常展示
AI 推荐以独立区域展示在关键词结果下方（虚线分隔）
       ↓
用户点击 AI 推荐 → 进入对应 Prompt 详情页
```

### 限流策略

| 条件 | 行为 |
|------|------|
| 关键词结果 ≥ 10 | 不触发 AI（已有足够结果） |
| 搜索词 < 4 字符 | 不触发（信息量不足） |
| 候选 Prompt > 50 | 不触发（token 消耗太大） |
| 候选 Prompt = 0 | 不触发（没什么可推荐的） |
| 两次搜索间隔 < 30s | 不触发（防连续调用浪费） |
| AI 未配置 | 不触发（静默跳过） |

const [isRunning, setIsRunning] = useState(false);

// 同时向选中的所有模型发送请求
const handleRunAll = async (content: string) => {
  setIsRunning(true);
  const items: ModelTestItem[] = selectedModels.map((m) => ({
    model: m,
    output: '',
    latency: 0,
    tokensUsed: 0,
    status: 'running' as const,
  }));
  setResults(items);

  // 并行请求
  await Promise.allSettled(
    selectedModels.map(async (model, i) => {
      const startTime = Date.now();
      try {
        const provider = getProviderForModel(model);
        if (!provider) throw new Error('Provider not available');

        const output = await provider.streamChat(
          [{ role: 'user', content }],
          (text) => {
            setResults((prev) => {
              const next = [...prev];
              next[i] = { ...next[i], output: next[i].output + text };
              return next;
            });
          }
        );

        setResults((prev) => {
          const next = [...prev];
          next[i] = {
            ...next[i],
            output,
            latency: Date.now() - startTime,
            status: 'done',
          };
          return next;
        });
      } catch (err) {
        setResults((prev) => {
          const next = [...prev];
          next[i] = {
            ...next[i],
            error: String(err),
            status: 'error',
            latency: Date.now() - startTime,
          };
          return next;
        });
      }
    })
  );

  setIsRunning(false);
};

## Step 15：Prompt 静态质量分析

### 概述

在 Prompt 编辑器中集成**本地静态分析引擎**，不依赖任何 AI API，完全通过算法和规则对 Prompt 内容进行质量评估。借鉴 `senior-prompt-engineer` 中 `prompt_optimizer.py` 的设计理念（520 行完整实现），将其中 token 估算、模糊指令检测、清晰度/结构评分、优化建议生成等能力引入产品。

与 P0 Step 4（AI 优化建议）的区别：

| 维度 | AI 优化建议（P0） | 静态质量分析（P2 Step 15） |
|------|-----------------|--------------------------|
| **引擎** | 调用 OpenAI/Anthropic API | 本地算法 + 正则 + 启发式规则 |
| **延迟** | 1-5 秒（网络 + 推理） | < 10ms（即时） |
| **成本** | 每次分析消耗 token | 零成本 |
| **离线** | 不可用 | 完全可用 |
| **输出** | AI 重写优化版本 | 质量诊断报告 + 改进建议 |
| **关系** | 生成新内容 | 发现问题，为 AI 优化提供方向 |

两者互补：先用静态分析快速发现问题，再决定是否调用 AI 进行深度优化。

### 静态分析引擎 — `src/services/ai/prompt-analyzer.ts`

```typescript
// ===== 分析引擎配置 =====

/** 各模型的字符/Token 比率（chars per token） */
const TOKEN_RATIOS: Record<string, number> = {
  'gpt-4o': 4.0,
  'gpt-4o-mini': 4.0,
  'gpt-4.1': 4.0,
  'gpt-4.1-mini': 4.0,
  'claude-sonnet-4-6': 3.5,
  'claude-haiku-4-5': 3.5,
};

/** 各模型每 1K token 成本（USD） */
const COST_PER_1K: Record<string, { input: number; output: number }> = {
  'gpt-4o':        { input: 0.0025, output: 0.01 },
  'gpt-4o-mini':   { input: 0.00015, output: 0.0006 },
  'gpt-4.1':       { input: 0.002, output: 0.008 },
  'gpt-4.1-mini':  { input: 0.0004, output: 0.0016 },
  'claude-sonnet-4-6': { input: 0.003, output: 0.015 },
  'claude-haiku-4-5':  { input: 0.001, output: 0.005 },
};

// ===== 模糊指令检测模式 =====

const AMBIGUOUS_PATTERNS: { regex: RegExp; type: ClarityIssue['type']; desc: string }[] = [
  { regex: /\b(etc|et cetera|and so on|and more|\.\.\.)\b/i, type: 'open-ended', desc: '使用了开放式列举词，应明确列举项' },
  { regex: /\b(various|multiple|several|many|different)\b/i, type: 'vague-verb', desc: '数量词不精确，应指定具体数量' },
  { regex: /\b(appropriate|suitable|proper|relevant)\b/i, type: 'subjective-term', desc: '主观判断词，缺乏客观标准' },
  { regex: /\b(as needed|if necessary|when appropriate|as applicable)\b/i, type: 'ambiguous-phrase', desc: '条件含糊，应明确触发条件' },
  { regex: /\b(maybe|perhaps|possibly|might|could)\b/i, type: 'vague-verb', desc: '语气不确定，应使用确定性表述' },
  { regex: /\b(for example|such as|e\.g\.|like)\b(?!.*\n)/i, type: 'open-ended', desc: '举例未完成或未限定范围' },
  { regex: /\b(optimize|improve|enhance|better)\b/i, type: 'vague-verb', desc: '动词过于笼统，应说明具体措施' },
  { regex: /\b(understand|analyze|consider|think about)\b/i, type: 'vague-verb', desc: '认知动词需要具体输出形式' },
  { regex: /\b(good|nice|high-quality|best|great)\b/i, type: 'subjective-term', desc: '主观形容词，应定义可衡量标准' },
  { regex: /\b(depending on|based on|according to)\b(?!\s+(the|an?|this|that|your))/i, type: 'ambiguous-phrase', desc: '依赖条件未指定' },
];

// ===== 主分析函数 =====

export function analyzePrompt(content: string): AnalysisReport {
  const clarity = analyzeClarity(content);
  const structure = analyzeStructure(content);
  const redundancy = analyzeRedundancy(content);
  const tokens = estimateTokens(content);

  const weighted = Math.round(
    clarity.score * 0.4 + structure.score * 0.35 + redundancy.score * 0.25
  );

  const suggestions = generateSuggestions(clarity, structure, redundancy, tokens);

  return {
    overall: { clarity: clarity.score, structure: structure.score, weighted },
    tokens,
    clarity,
    structure,
    redundancy,
    suggestions,
    timestamp: Date.now(),
  };
}

// ===== Token 估算 =====

function estimateTokens(content: string): TokenEstimate[] {
  const charCount = content.length;
  return Object.entries(TOKEN_RATIOS).map(([model, ratio]) => {
    const tokens = Math.round(charCount / ratio);
    const costPer1K = COST_PER_1K[model] ?? { input: 0.003, output: 0.015 };
    return {
      model,
      modelLabel: model,
      ratio,
      inputChars: charCount,
      estimatedTokens: tokens,
      costPer1KInput: costPer1K.input,
      costPer1KOutput: costPer1K.output,
      estimatedInputCost: (tokens / 1000) * costPer1K.input,
      estimatedOutputCost: (tokens / 1000) * costPer1K.output,
    };
  });
}

// ===== 清晰度分析 =====

function analyzeClarity(content: string): ClarityAnalysis {
  const issues: ClarityIssue[] = [];

  for (const pattern of AMBIGUOUS_PATTERNS) {
    const matches = content.matchAll(pattern.regex);
    for (const match of matches) {
      if (match.index !== undefined) {
        issues.push({
          type: pattern.type,
          text: match[0],
          description: pattern.desc,
          suggestion: `将"${match[0]}"替换为更精确的表述`,
          position: { start: match.index, end: match.index + match[0].length },
        });
      }
    }
  }

  const hasOutputFormat = /(json|xml|csv|yaml|markdown|表格|列表|段落)/i.test(content) &&
    /(返回|输出|格式|以.*形式|format|output|return)/i.test(content);

  const hasDirectives = /(请|必须|需要|应当|不要|避免|规则|要求|步骤|条件|准则|rule|requirement|step|condition)/i.test(content);

  const ambiguityDeductions = issues.filter(i => ['vague-verb', 'subjective-term', 'ambiguous-phrase'].includes(i.type)).length * 5;
  const openEndedDeductions = issues.filter(i => i.type === 'open-ended').length * 4;

  let score = 100;
  score -= Math.min(ambiguityDeductions, 40);
  score -= Math.min(openEndedDeductions, 20);
  if (!hasOutputFormat) score -= 10;
  if (!hasDirectives) score -= 5;
  score = Math.max(0, score);

  return {
    score,
    issues,
    totalDeductions: 100 - score,
    hasOutputFormat,
    hasDirectives,
  };
}

// ===== 结构分析 =====

function analyzeStructure(content: string): StructureAnalysis {
  const sections: StructureSection[] = [];

  // Markdown 标题
  const headerRegex = /^(#{1,3})\s+(.+)$/gm;
  let match;
  while ((match = headerRegex.exec(content)) !== null) {
    sections.push({
      name: match[2].trim(),
      type: 'markdown-header',
      content: match[2].trim(),
      charCount: match[2].trim().length,
    });
  }

  // 常见标签名（大写开头标签如 "Role:", "Goals:" 等）
  const labelRegex = /^([A-Z][A-Za-z\s]+)[：:]/gm;
  while ((match = labelRegex.exec(content)) !== null) {
    const name = match[1].trim();
    if (!sections.some(s => s.name === name)) {
      sections.push({ name, type: 'label', content: name, charCount: name.length });
    }
  }

  // 常见章节名（英文）
  const commonNames = ['role', 'goal', 'task', 'context', 'constraint', 'example', 'output', 'format', 'rule', 'step', 'instruction'];
  for (const name of commonNames) {
    const regex = new RegExp(`\\b${name}\\b`, 'i');
    if (regex.test(content) && !sections.some(s => s.name.toLowerCase() === name)) {
      sections.push({ name, type: 'common-name', content: name, charCount: name.length });
    }
  }

  // Few-shot 示例检测
  const fewShotCount = countFewShotExamples(content);

  // 结构评分
  let score = 50;
  if (sections.length >= 2) score += 15;
  if (sections.length >= 4) score += 10;
  if (hasOutputFormatSpec(content)) score += 15;
  if (fewShotCount > 0) score += 10;
  score = Math.min(100, score);

  return {
    score,
    sections,
    totalSections: sections.length,
    hasExamples: fewShotCount > 0,
    fewShotCount,
  };
}

function countFewShotExamples(content: string): number {
  let count = 0;

  // 模式 1: "Example:" / "示例：" 标记
  const exampleLabels = content.match(/^(Example|示例|例子|Sample|Input|输出示例)[：:]/gm);
  if (exampleLabels) count += exampleLabels.length;

  // 模式 2: 用户/助手对话分隔（``` 包裹的对话块）
  const dialogBlocks = content.match(/(User|Human|Assistant|AI)[：:][\s\S]*?(?=\n\n|$)/g);
  if (dialogBlocks) count += Math.min(dialogBlocks.length, 5);

  return count;
}

function hasOutputFormatSpec(content: string): boolean {
  return /(output|输出|return|返回|format|格式|response|响应)/i.test(content);
}

// ===== 冗余检测 =====

function analyzeRedundancy(content: string): RedundancyAnalysis {
  const issues: RedundancyIssue[] = [];
  const lines = content.split('\n').filter(l => l.trim().length > 10);
  const checked = new Set<string>();

  for (let i = 0; i < lines.length; i++) {
    const words = lines[i].trim().split(/\s+/);
    if (words.length < 5) continue;

    // 检查 3 词以上的短语是否在其他行重复
    for (let phraseLen = 3; phraseLen <= Math.min(6, words.length); phraseLen++) {
      for (let start = 0; start <= words.length - phraseLen; start++) {
        const phrase = words.slice(start, start + phraseLen).join(' ').toLowerCase();
        if (phrase.length < 10 || checked.has(phrase)) continue;
        checked.add(phrase);

        // 在其他行搜索
        let occurrences = 0;
        const positions: { start: number; end: number }[] = [];
        const searchRegex = new RegExp(phrase.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
        let m;
        while ((m = searchRegex.exec(content)) !== null) {
          occurrences++;
          positions.push({ start: m.index, end: m.index + m[0].length });
        }

        if (occurrences >= 2) {
          issues.push({
            repeatedPhrase: phrase,
            occurrences,
            positions,
            suggestion: `短语"${phrase}"重复出现 ${occurrences} 次，考虑合并或提取为变量`,
          });
        }
      }
    }

    // 限制检测量，避免性能问题
    if (issues.length >= 10) break;
  }

  const redundancies = issues.reduce((sum, i) => sum + i.occurrences, 0);
  const deduction = Math.min(redundancies * 3, 30);
  const score = Math.max(70, 100 - deduction);

  return { score, issues, totalRedundancies: issues.length };
}

// ===== 建议生成 =====

function generateSuggestions(
  clarity: ClarityAnalysis,
  structure: StructureAnalysis,
  redundancy: RedundancyAnalysis,
  tokens: TokenEstimate[]
): Suggestion[] {
  const suggestions: Suggestion[] = [];

  // 规则 1: 输出格式缺失
  if (!clarity.hasOutputFormat) {
    suggestions.push({
      severity: 'warning',
      category: 'format',
      title: '缺少输出格式说明',
      description: '建议在 Prompt 中指定输出格式（JSON/XML/Markdown 等），提高输出一致性',
    });
  }

  // 规则 2: 缺少指令
  if (!clarity.hasDirectives) {
    suggestions.push({
      severity: 'warning',
      category: 'clarity',
      title: '缺少明确指令规则',
      description: '添加"必须""避免""规则"等约束性表述，减少模型自由度',
    });
  }

  // 规则 3: 结构过于简单
  if (structure.totalSections < 2) {
    suggestions.push({
      severity: 'warning',
      category: 'structure',
      title: 'Prompt 结构单一',
      description: '使用 Markdown 标题或标签将 Prompt 划分为多个段落（角色、目标、约束、输出格式）',
    });
  }

  // 规则 4: 模糊指令
  if (clarity.issues.length > 0) {
    suggestions.push({
      severity: clarity.issues.length > 3 ? 'error' : 'warning',
      category: 'clarity',
      title: `发现 ${clarity.issues.length} 个模糊表述`,
      description: `使用精确数量、客观标准代替"很多""合适""等模糊词汇`,
    });
  }

  // 规则 5: 缺少示例
  if (!structure.hasExamples) {
    suggestions.push({
      severity: 'info',
      category: 'examples',
      title: '建议添加 Few-shot 示例',
      description: '提供 1-3 个输入/输出示例，可显著提升模型对格式和内容的遵循度',
    });
  }

  // 规则 6: 存在冗余
  if (redundancy.totalRedundancies > 0) {
    suggestions.push({
      severity: redundancy.totalRedundancies > 3 ? 'warning' : 'info',
      category: 'clarity',
      title: `发现重复短语（${redundancy.totalRedundancies} 处）`,
      description: redundancy.issues[0]?.suggestion ?? '合并重复表述，精简 Prompt 长度',
    });
  }

  // 规则 7: Token 成本警告
  const maxTokens = Math.max(...tokens.map(t => t.estimatedTokens));
  if (maxTokens > 4000) {
    suggestions.push({
      severity: 'warning',
      category: 'token',
      title: `Prompt 较长（约 ${maxTokens} tokens）`,
      description: `每次调用约 $${(maxTokens / 1000 * 0.01).toFixed(4)}（Claude Sonnet 输出），建议精简`,
    });
  } else if (maxTokens < 50) {
    suggestions.push({
      severity: 'info',
      category: 'token',
      title: 'Prompt 较短',
      description: '考虑补充上下文和约束条件以提高输出质量',
    });
  }

  // 规则 8: 示例数量优化
  if (structure.fewShotCount > 5) {
    suggestions.push({
      severity: 'info',
      category: 'examples',
      title: `示例数量较多（${structure.fewShotCount} 个）`,
      description: '超过 5 个示例可能边际收益递减且增加 token 消耗，建议精选 2-3 个最有代表性的',
    });
  }

  return suggestions;
}

export { analyzePrompt as default };
```

## Step 15 专家评审结论

### 评审背景

由三位专家（AI 产品专家、前端架构师、产品经理）对 Step 15 Prompt 静态质量分析进行了独立评估。以下是综合结论。

### 评审结论总览

| 维度 | AI 产品专家 | 前端架构师 | 产品经理 |
|------|-----------|-----------|---------|
| **用户价值** | 高 — 填补"不知好不好"的认知断层 | — | 真实但被动，需正确触发 |
| **可行性** | 高，风险极低 | 高，注意 ReDoS 和中文估算偏差 | — |
| **竞品差异化** | 明确生态位，暂无同类产品 | — | 护城河浅，竞品可快速复刻 |
| **推荐优先级** | P2 | P1（提前） | P2 |
| **开发成本** | ~400 行，1-2 天 | ~450 行，中等复杂度 | — |

### 关键结论：维持 P2，调整实现策略

#### 1. 产品形态：从独立功能改为嵌入 AI 优化流程

三位专家一致认为**不应做成独立面板让用户主动点击**。产品经理给出关键策略：

```
用户点击"AI 优化"按钮
       ↓
先触发静态分析（免费，<100ms）
       ↓
展示分析报告 + quick win 建议
       ↓
用户再决定是否花 token 调 API 做 AI 重写
       ↓
静态分析发现问题 → AI 针对性优化（串联）
```

这样做的好处：
- 解决"用户不会主动用"的问题，分析结果自动出现
- 为 P0 AI 优化做流量铺垫，降低用户对"花 token"的心理门槛
- 兜住"不想花 token"的用户，至少提供免费的质量反馈

#### 2. 代码调整

根据前端架构师评审，对已规划代码做以下修正：

| 原规划 | 调整后 | 理由 |
|--------|--------|------|
| `src/services/ai/prompt-analyzer.ts` | `src/services/promptAnalyzer.ts` | 功能纯本地不调用 AI，去掉 `ai/` 目录 |
| `src/types/analysis.ts`（新建） | 合并到 `types/index.ts` | 维持现有惯例，当前类型文件仅 44 行，拆分时机未到 |
| 引擎为单一函数 | 采用 RuleSet 模式 | 每条规则独立实现 `detect()` + `suggest()`，新增规则无需改主函数 |
| 评分权重硬编码 | 提取为常量 `SCORE_WEIGHTS` | 便于调优和扩展 |
| — | 添加正则回溯安全检查 | 超长输入（>5000 字符）跳过部分检查，防止 ReDoS |
| Token 估算仅英文比率 | 增加中文 char/token 比率区分 | 中英文混合 Prompt 下估算偏差可缩小 |
| 结果不缓存 | 内存缓存（`useRef`） | 切换 tab 再切回时避免重复计算 |
| 性能标注 `<10ms` | 标注"通常在 10ms 内，1万字场景不超过 100ms" | 更准确反映边缘场景 |

  const isLongInput = content.length > MAX_SAFE_LENGTH;

  const patterns = isLongInput
    ? AMBIGUOUS_PATTERNS.filter(p => p.type !== 'vague-verb') // 跳过回溯风险高的模式
    : AMBIGUOUS_PATTERNS;

  // ...
}

// 每条正则标注 ReDoS 风险等级
const AMBIGUOUS_PATTERNS = [
  {
    regex: /\b(various|multiple|several)\b/i,
    type: 'vague-verb' as const,
    desc: '...',
    redoRisk: 'low' as const,    // 确定性有限长度，安全
  },
  // ...
  {
    regex: /\b(depending on|based on|according to)\b(?!\s+(the|an?|this|that|your))/i,
    type: 'ambiguous-phrase' as const,
    desc: '...',
    redoRisk: 'medium' as const, // 含否定前瞻，但无嵌套量词
  },
];
```

#### 5. Token 估算增加中文区分

```typescript
const TOKEN_RATIOS: Record<string, number> = {
  'gpt-4o': 4.0,          // 英文 chars/token
  'claude-sonnet-4-6': 3.5,
};

// 中文的 char/token 比率约为英文的一半
// 检测中文字符占比，加权计算估算值
function estimateTokensForModel(content: string, model: string): number {
  const ratio = TOKEN_RATIOS[model] ?? 4.0;
  const chineseRatio = (content.match(/[一-鿿]/g)?.length ?? 0) / content.length;

  // 中文占比越高，effectiveRatio 越低（更多 chars per token）
  const effectiveRatio = ratio * (1 - chineseRatio * 0.4);
  return Math.round(content.length / effectiveRatio);
}
```

#### 6. 优先级最终决定：P2
