### AI Service 统一入口 — `src/services/ai/index.ts`

```typescript
import type { AIProvider } from './provider';
import { OpenAIProvider } from './openai';
import { AnthropicProvider } from './anthropic';
import type { AIProviderType, AIOptimizeRequest, AIStreamChunk } from '@/types/ai';

let currentProvider: AIProvider | null = null;

export function getAIProvider(type: AIProviderType, apiKey: string, model: string): AIProvider {
  switch (type) {
    case 'openai': return new OpenAIProvider(apiKey, model);
    case 'anthropic': return new AnthropicProvider(apiKey, model);
  }
}

export function initAI(settings: { type: AIProviderType; apiKey: string; model: string }) {
  currentProvider = getAIProvider(settings.type, settings.apiKey, settings.model);
}

export function getCurrentProvider(): AIProvider | null {
  return currentProvider;
}

// 优化 Prompt
export async function optimizePrompt(
  request: AIOptimizeRequest,
  onChunk?: (chunk: AIStreamChunk) => void,
  signal?: AbortSignal
): Promise<string> {
  const provider = getCurrentProvider();
  if (!provider) throw new Error('AI 未配置，请在设置中配置 API Key');

  const systemPrompt = `你是一个 Prompt 工程专家。请优化用户提供的 Prompt，使其更清晰、有效。\
仅返回优化后的 Prompt 内容，不要添加解释。考虑以下方面：
- 指令清晰度
- 角色设定完整性
- 输出格式引导
- 约束条件明确度`;

  const messages = [
    { role: 'system' as const, content: systemPrompt },
    ...(request.context ? [{ role: 'user' as const, content: request.context }] : []),
    { role: 'user' as const, content: request.content },
    ...(request.instruction ? [{ role: 'user' as const, content: `额外要求：${request.instruction}` }] : []),
  ];

  if (onChunk) {
    return provider.streamChat(messages, (text) => {
      onChunk({ type: 'text', content: text });
    }, signal);
  }
  return provider.chat(messages, signal);
}

// 生成 Prompt
export async function generatePrompt(
  description: string,
  onChunk?: (text: string) => void,
  signal?: AbortSignal
): Promise<string> {
  const provider = getCurrentProvider();
  if (!provider) throw new Error('AI 未配置');

  const systemPrompt = '你是一个 Prompt 生成专家。根据用户的需求描述，生成一个高质量的 Prompt。\
确保包含角色设定、任务描述、输出格式等要素。直接返回 Prompt 内容。';

  const messages = [{ role: 'system', content: systemPrompt }, { role: 'user', content: description }];

  if (onChunk) {
    return provider.streamChat(messages, onChunk, signal);
  }
  return provider.chat(messages, signal);
}
```

### useVariables Hook — `src/hooks/useVariables.ts`

```typescript
import { useMemo } from 'react';

export interface VariableInfo {
  name: string;
  value: string;
}

export function extractVariables(content: string): string[] {
  const matches = content.match(/\{\{(\w+)\}\}/g);
  if (!matches) return [];
  return [...new Set(matches.map((m) => m.slice(2, -2)))];
}

export function renderTemplate(template: string, variables: VariableInfo[]): string {
  let result = template;
  for (const v of variables) {
    result = result.replace(new RegExp(`\\{\\{${v.name}\\}\\}`, 'g'), v.value);
  }
  return result;
}

export function useVariables(content: string) {
  return useMemo(() => {
    const names = extractVariables(content);
    return {
      hasVariables: names.length > 0,
      variableNames: names,
    };
  }, [content]);
}
```

```typescript
const [isOptimizing, setIsOptimizing] = useState(false);
const [result, setResult] = useState<string | null>(null);
const [instruction, setInstruction] = useState('');
const [error, setError] = useState<string | null>(null);
const abortRef = useRef<AbortController | null>(null);

const handleOptimize = async () => {
  setIsOptimizing(true);
  setError(null);
  const controller = new AbortController();
  abortRef.current = controller;

  try {
    let accumulated = '';
    await optimizePrompt(
      { content, instruction: instruction || undefined },
      (chunk) => {
        if (chunk.type === 'text') {
          accumulated += chunk.content;
          // 流式更新结果（逐字显示）
          setResult(accumulated);
        } else if (chunk.type === 'error') {
          setError(chunk.content);
        }
      },
      controller.signal
    );
  } catch (err: any) {
    if (err.name !== 'AbortError') {
      setError(err.message || '优化请求失败');
    }
  } finally {
    setIsOptimizing(false);
    abortRef.current = null;
  }
};
```

```typescript
import { computeDiff, diffToLines } from '@/utils/diff';

interface OptimizeDiffProps {
  original: string;
  optimized: string;
}

export default function OptimizeDiff({ original, optimized }: OptimizeDiffProps) {
  const diffs = computeDiff(original, optimized);
  const lines = diffToLines(diffs);

  return (
    <div className="border rounded-md overflow-hidden font-mono text-sm">
      {lines.map((line, i) => (
        <div key={i} className={`px-3 py-0.5 ${
          line.type === 'insert' ? 'bg-green-500/10 text-green-600 dark:text-green-400' :
          line.type === 'delete' ? 'bg-red-500/10 text-red-600 dark:text-red-400 line-through' :
          'text-foreground'
        }`}>
          <span className="mr-2 select-none text-muted-foreground">
            {line.type === 'insert' ? '+' : line.type === 'delete' ? '-' : ' '}
          </span>
          {line.text}
        </div>
      ))}
    </div>
  );
}
```

```typescript
const abortRef = useRef<AbortController | null>(null);

const handleRunTest = async () => {
  setIsRunning(true);
  setOutput('');
  const controller = new AbortController();
  abortRef.current = controller;

  const provider = getCurrentProvider();
  if (!provider) { /* 提示配置 API Key */ return; }

  try {
    const fullOutput = await provider.streamChat(
      [{ role: 'user', content }],
      (text) => setOutput((prev) => prev + text),
      controller.signal
    );

    // 保存测试输出到版本记录
    await setVersionTestOutput(versionId, fullOutput, model);
  } catch (err: any) {
    if (err.name !== 'AbortError') {
      setOutput(`错误：${err.message}`);
    }
  } finally {
    setIsRunning(false);
    abortRef.current = null;
  }
};
```

## 关键设计决策

### 1. API Key 存储

前端直连方案：Key 存储在 `localStorage`，不经过任何后端。

- **为什么可以接受**：TypingMind、ChatGPT Next、Open WebUI 等主流工具都采用此方案
- **用户预期**：用户输入自己的 Key，自己承担管理责任
- **替代路径**：未来可加服务端代理（第二期），Key 转移到后端环境变量

### 2. Provider 插件化

```typescript
// 新增 provider 只需实现 AIProvider 接口
export class OllamaProvider implements AIProvider { ... }
export class GeminiProvider implements AIProvider { ... }
```

新 provider 只需实现 `streamChat` 和 `chat` 两个方法，在 `ai/index.ts` 注册即可。

### 3. 变量模板与版本管理的关系

- 保存的是**原始模板**（含 `{{变量}}` 标记），保证版本历史可读
- `variables[]` 字段记录变量名列表，用于快速识别
- 复制/测试时使用**渲染后内容**（变量值替换后）

### 4. 评分与版本号的关系

- 评分不参与版本号生成（版本号仍然是 patch 递增）
- 评分是版本的元数据，影响列表排序但不影响版本标识
- 评分可随时修改，不触发新版本创建
